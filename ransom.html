/* --- TYPE 1 --- */ if (popupType === 1) { randomPopupPosition(); text.textContent = randomText(type1); popupVariant = 1; } /* --- TYPE 2 --- */ else if (popupType === 2) { randomPopupPosition(); text.textContent = randomText(type2);
